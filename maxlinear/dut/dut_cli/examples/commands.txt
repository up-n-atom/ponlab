# Example of batch file for the DUT Command Line Interface.
# Run this file by issuing the 'load' command at the dut_cli prompt.

# This series of commands will initialize and configure an LGM to transmit 
# packets using the 5GHz band.

exec 2 driverInit --memory-type 2 --memory-size 1024

exec 2 setChannel --phy-mode 7 --bandwidth 0 --channel 36

exec 2 setRate --bandwidth 0 --mcs 2 --nss 1 --gi 0 --ltf 1 

exec 2 setClosedLoopEnabled --enabled

exec 2 setTransmitPowerLimit --power-limit 15

exec 2 startTx --repetitions 65535 --packet-length 1000

sleep 2000

exec 2 stopTx

exec 2 driverRelease
